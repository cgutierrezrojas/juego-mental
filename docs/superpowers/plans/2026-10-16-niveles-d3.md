# Juego Mental — Plan D3: niveles ampliados

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** Implementar `docs/superpowers/specs/2026-10-16-niveles-d3-design.md`: 50 niveles en 5 mundos para los 12 juegos, estrellas 1–3 y retos (Sin fallar, Contrarreloj, Jefe).

**Arquitectura:** `games/niveles.js` se reescribe (puro): configuración de cada nivel con sus 3 metas y su reto, cálculo de estrellas y progreso. Los juegos ya no terminan al llegar a la meta. Terminan al llegar a `nivel.tope` (⭐⭐⭐), y con `nivel.sinFallar` el primer fallo termina. `main.js` guarda las estrellas en `estrellas:<id>` (con el progreso antiguo como 1 ⭐) y dibuja mundos, niveles y el final con estrellas.

**Tecnología:** HTML + CSS + JavaScript (módulos ES), sin frameworks ni compilación.

## Restricciones globales

- Sin frameworks, sin dependencias, sin paso de compilación. Código simple y legible; nombres en español.
- Los módulos de `games/` no tocan `document`, `localStorage`, `navigator` ni `AudioContext` al cargarse.
- Fuera del modo Niveles nada cambia.
- `start` devuelve `parar()`; tras `parar()` no se llama a `alTerminar` ni queda nada programado.
- Valores exactos: los del diseño (fórmulas, retos, estrellas ×1,25 / ×1,5 y ×0,9 / ×0,8).

```js
// Niveles: 50 por juego en 5 mundos de 10, con estrellas (1–3) y retos. Todo puro.

export const NUM_NIVELES = 50;
export const NOMBRE_RETO = { normal: '⏱ Normal', sinfallo: '❤️ Sin fallar', contrarreloj: '⚡ Contrarreloj', jefe: '👑 Jefe' };
export const ICONO_RETO = { normal: '', sinfallo: '❤️', contrarreloj: '⚡', jefe: '👑' };

// Segundos de los juegos de opciones; Atención y Topos son de 30 s.
const OPCIONES = { calculo: 30, mayor: 30, distinto: 30, series: 60, sobra: 60, balanza: 60 };
const CON_TIEMPO = [...Object.keys(OPCIONES), 'atencion', 'topos'];
const MENOR_ES_MEJOR = ['puzzle', 'rayo', 'parejas'];

// Reto del nivel n: jefe cada 10, sin fallar cada 5 (sin contar los jefes), contrarreloj en los terminados en 3 y 7.
export function retoDe(n) {
  if (n % 10 === 0) return 'jefe';
  if (n % 5 === 0) return 'sinfallo';
  if (n % 10 === 3 || n % 10 === 7) return 'contrarreloj';
  return 'normal';
}

// Meta base (sin reto) y parámetros del juego en el nivel n.
function base(id, n) {
  if (id in OPCIONES) {
    return { meta: 5 + Math.round(n * 0.3), segundos: OPCIONES[id], aciertosIniciales: Math.min(n - 1, 20), dificultad: 'normal', unidad: 'aciertos' };
  }
  if (id === 'atencion') return { meta: 6 + Math.round(n * 0.35), segundos: 30, dificultad: n >= 11 ? 'dificil' : 'normal', unidad: 'puntos' };
  if (id === 'memoria') return { meta: 3 + Math.floor(n * 0.25), dificultad: n >= 21 ? 'dificil' : 'normal', unidad: 'colores' };
  if (id === 'topos') return { meta: 10 + Math.round(n * 0.3), segundos: 30, dificultad: n <= 10 ? 'facil' : n <= 30 ? 'normal' : 'dificil', unidad: 'puntos' };
  if (id === 'puzzle') return { meta: Math.round(25 + n * 1.2), desorden: 10 + n * 4, unidad: 'movimientos' };
  if (id === 'rayo') return { meta: 450 - n * 4, unidad: 'ms' };
  if (id === 'parejas') {
    const parejas = 3 + Math.ceil(n / 10); // 4 parejas en el mundo 1 … 8 en el 5
    return { meta: Math.round(parejas * 2.2), parejas, unidad: 'intentos' };
  }
  return null;
}

// Configuración completa del nivel n de un juego: base + reto + las 3 metas de las estrellas.
export function configNivel(id, n) {
  const b = base(id, n);
  if (!b) return null;
  const reto = retoDe(n);
  const menor = MENOR_ES_MEJOR.includes(id);
  const conTiempo = CON_TIEMPO.includes(id);
  let { meta, segundos = 0, dificultad = 'normal' } = b;
  let sinFallar = false;
  if (reto === 'jefe') {
    meta = Math.round(meta * (menor ? 0.9 : 1.1));
  } else if (reto === 'sinfallo' && conTiempo) {
    sinFallar = true;
    meta = Math.round(meta * 0.7);
  } else if (reto === 'contrarreloj') {
    if (conTiempo) segundos = Math.round(segundos * 0.75);
    else if (id === 'memoria') dificultad = 'dificil';
  }
  const metas = menor
    ? [meta, Math.round(meta * 0.9), Math.round(meta * 0.8)]
    : [meta, Math.ceil(meta * 1.25), Math.ceil(meta * 1.5)];
  const signo = menor ? '≤' : '';
  const tiempo = segundos ? ` en ${segundos} s` : '';
  const texto = `⭐ ${signo}${metas[0]} · ⭐⭐ ${signo}${metas[1]} · ⭐⭐⭐ ${signo}${metas[2]} ${b.unidad}${tiempo}`;
  return { aciertosIniciales: 0, ...b, meta, metas, tope: metas[2], segundos, dificultad, sinFallar, reto, texto };
}

// Estrellas conseguidas (0–3) con una puntuación. En "menos es mejor", 0 no es una marca válida.
export function estrellasDe(puntos, metas, menorEsMejor = false) {
  if (menorEsMejor) return puntos > 0 ? metas.filter((m) => puntos <= m).length : 0;
  return metas.filter((m) => puntos >= m).length;
}

// `estrellas`: lista de 50 números 0–3 (índice n − 1). Un nivel está libre si es el 1 o el anterior tiene ⭐.
export const nivelLibre = (estrellas, n) => n === 1 || (estrellas[n - 2] ?? 0) > 0;

// Guarda la mejor marca del nivel n (nunca baja). Devuelve una lista nueva.
export function conEstrellas(estrellas, n, nuevas) {
  const resultado = [...estrellas];
  resultado[n - 1] = Math.max(resultado[n - 1] ?? 0, nuevas);
  return resultado;
}

export const totalEstrellas = (estrellas) => estrellas.reduce((suma, e) => suma + (e || 0), 0);

// Progreso de la versión anterior (`superado` = nivel más alto superado) como 1 ⭐ por nivel.
export const desdeAntiguo = (superado) => Array.from({ length: NUM_NIVELES }, (_, i) => (i < superado ? 1 : 0));

// Primer nivel libre sin estrellas (o el último, si están todos).
export function siguienteNivel(estrellas) {
  for (let n = 1; n <= NUM_NIVELES; n++) if (nivelLibre(estrellas, n) && !(estrellas[n - 1] > 0)) return n;
  return NUM_NIVELES;
}
```

function leerEstrellas(id) {
  try {
    const estrellas = JSON.parse(leer('estrellas:' + id));

- [ ] **Step 4: Mundos y niveles con estrellas (`main.js`)**

Sustituir la función `pintarNiveles` entera (desde su comentario `// Cuadrícula de niveles (⭐ superados, 🔒 bloqueados) y el objetivo del nivel elegido.` hasta su `}` final) por:

```js
// Modo Niveles: total de estrellas, mundos (de 10 niveles), niveles del mundo con su reto y estrellas, y el objetivo.
function pintarNiveles(j) {
  const estrellas = leerEstrellas(j.id);
  const mundo = Math.ceil(nivelElegido / 10);
  const grupo = (titulo, clase) => {
    const caja = el('div', clase);
    caja.setAttribute('role', 'group');
    caja.setAttribute('aria-label', titulo);
    return caja;
  };
  // Al elegir, se repinta y el foco vuelve al botón elegido de esa fila.
  const elegirNivel = (n, fila) => {
    nivelElegido = n;
    pintarNiveles(j);
    $('previa-niveles').querySelector(`[aria-label="${fila}"] [aria-pressed="true"]`)?.focus();
  };

  const mundos = grupo('Mundo', 'mundos');
  for (let m = 1; m <= NUM_NIVELES / 10; m++) {
    const primero = (m - 1) * 10 + 1;
    const libre = nivelLibre(estrellas, primero);
    const completo = estrellas.slice(primero - 1, primero + 9).every((e) => e > 0);
    const b = el('button', '', `${libre ? '' : '🔒'}${m}${completo ? ' ✓' : ''}`);
    b.disabled = !libre;
    b.setAttribute('aria-pressed', m === mundo);
    b.setAttribute('aria-label', `Mundo ${m}${completo ? ', completo' : libre ? '' : ', bloqueado'}`);
    // Al cambiar de mundo se elige su primer nivel libre sin estrellas (o el primero del mundo).
    b.onclick = () => {
      let destino = primero;
      for (let n = primero + 9; n >= primero; n--) if (nivelLibre(estrellas, n) && !(estrellas[n - 1] > 0)) destino = n;
      elegirNivel(destino, 'Mundo');
    };
    mundos.append(b);
  }

  const rejilla = grupo('Nivel', 'niveles');
  for (let n = (mundo - 1) * 10 + 1; n <= mundo * 10; n++) {
    const libre = nivelLibre(estrellas, n);
    const e = estrellas[n - 1] || 0;
    const reto = retoDe(n);
    const b = el('button', reto === 'normal' ? '' : 'reto');
    b.append(el('span', '', libre ? `${n}` : `🔒${n}`), el('small', '', libre ? `${ICONO_RETO[reto]}${'⭐'.repeat(e)}${'☆'.repeat(3 - e)}` : ''));
    b.disabled = !libre;
    b.setAttribute('aria-pressed', n === nivelElegido);
    b.setAttribute('aria-label', libre ? `Nivel ${n}, ${NOMBRE_RETO[reto]}, ${e} de 3 estrellas` : `Nivel ${n}, bloqueado`);
    b.onclick = () => elegirNivel(n, 'Nivel');
    rejilla.append(b);
  }

  const cfg = configNivel(j.id, nivelElegido);
  const objetivo = el('p', 'objetivo', `Nivel ${nivelElegido} · ${NOMBRE_RETO[cfg.reto]}\n${cfg.texto}`);
  objetivo.setAttribute('aria-live', 'polite'); // se anuncia al cambiar de nivel
  const total = el('p', 'total-estrellas', `⭐ ${totalEstrellas(estrellas)}/${NUM_NIVELES * 3}`);
  $('previa-niveles').replaceChildren(total, mundos, rejilla, objetivo);
  $('btn-jugar').textContent = `▶ Jugar nivel ${nivelElegido}`;
}
```
