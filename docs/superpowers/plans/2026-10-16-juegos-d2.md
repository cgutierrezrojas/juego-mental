# Juego Mental — Plan D2: cuatro juegos nuevos

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** Implementar `docs/superpowers/specs/2026-10-16-juegos-d2-design.md`: ⚖️ Mayor o menor, 🎯 El distinto, 🏋️ Balanza y 🃏 Parejas.

**Arquitectura:** Mayor o menor, El distinto y Balanza reutilizan `jugarConOpciones` (modos, dificultad, reloj y `parar`), que gana dos campos opcionales en la ronda: `columnas` y `pintar(boton, valor)`. Parejas es un módulo propio. Cada juego exporta su lógica pura para `tests.js`, tiene su entrada en `JUEGOS` (sin `categoria`, para que no entre en el entrenamiento diario) y su línea en `sw.js`.

**Tecnología:** HTML + CSS + JavaScript (módulos ES), sin frameworks ni compilación.

## Restricciones globales

- Sin frameworks, sin dependencias, sin paso de compilación. Código simple y legible; nombres en español.
- Los módulos de `games/` no tocan `document`, `localStorage`, `navigator` ni `AudioContext` al cargarse.
- `start(pantalla, alTerminar, opciones)` devuelve `parar()`; tras `parar()` no se llama a `alTerminar` ni queda nada programado.
- Los juegos nuevos **no tienen `categoria`**, así que no entran en el entrenamiento diario.
- Valores exactos: los del diseño.

// Mayor o menor: toca la operación que da más. 60 segundos, como Cálculo.
import { generarOperacion } from './calculo.js';
import { jugarConOpciones } from './opciones.js';

// Dos operaciones del tramo de dificultad que toque, que nunca dan lo mismo.
export function generarPar(aciertos, rnd = Math.random) {
  let a;
  let b;
  do {
    a = generarOperacion(aciertos, rnd);
    b = generarOperacion(aciertos, rnd);
  } while (a.resultado === b.resultado);
  return { a, b, mayor: a.resultado > b.resultado ? a.texto : b.texto };
}

export function start(pantalla, alTerminar, opciones = {}) {
  return jugarConOpciones(pantalla, alTerminar, (aciertos) => {
    const { a, b, mayor } = generarPar(aciertos);
    return { texto: '¿Cuál da más?', opciones: [a.texto, b.texto], correcta: mayor, columnas: 1 };
  }, '', opciones);
}
```

// El distinto: toca el cuadrado de color diferente. 60 segundos, como Cálculo.
import { azar } from './comun.js';
import { jugarConOpciones } from './opciones.js';

// Una ronda: lado de la cuadrícula, casilla distinta y los dos colores.
// Con los aciertos crece la cuadrícula y la diferencia de luminosidad es más sutil.
export function generarRonda(aciertos, rnd = Math.random) {
  const lado = aciertos < 4 ? 2 : aciertos < 9 ? 3 : aciertos < 15 ? 4 : 5;
  const diferencia = Math.max(4, 22 - aciertos);
  const tono = azar(360, rnd);
  const luz = 45 + azar(15, rnd);
  const signo = rnd() < 0.5 ? -1 : 1;
  return {
    lado,
    diferencia,
    distinto: azar(lado * lado, rnd),
    color: `hsl(${tono} 65% ${luz}%)`,
    colorDistinto: `hsl(${tono} 65% ${luz + signo * diferencia}%)`,
  };
}

export function start(pantalla, alTerminar, opciones = {}) {
  return jugarConOpciones(pantalla, alTerminar, (aciertos) => {
    const r = generarRonda(aciertos);
    return {
      texto: 'Toca el distinto',
      opciones: Array.from({ length: r.lado * r.lado }, (_, i) => i),
      correcta: r.distinto,
      columnas: r.lado,
      pintar: (boton, i) => {
        boton.textContent = '';
        boton.className = 'casilla-color';
        boton.style.background = i === r.distinto ? r.colorDistinto : r.color;
        boton.ariaLabel = `Casilla ${i + 1}`;
      },
    };
  }, 'distinto', opciones);
}
```

// Balanza: deduce cuánto pesa la fruta que se pregunta. 60 segundos, como Cálculo.
import { barajar, entre } from './comun.js';
import { generarOpciones, jugarConOpciones } from './opciones.js';

const FRUTAS = ['🍎', '🍌', '🍇', '🍐', '🍊', '🍓'];

// Ecuaciones con pesos ocultos (1–9), la fruta que se pregunta y su peso. Tramos según los aciertos:
// 1 ecuación (k frutas iguales), 2 (A A, A B → B) o 3 (A A, A B, B C → C).
export function generarBalanza(aciertos, rnd = Math.random) {
  const [A, B, C] = barajar(FRUTAS, rnd);
  const pesos = { [A]: entre(1, 9, rnd), [B]: entre(1, 9, rnd), [C]: entre(1, 9, rnd) };
  const ecuacion = (...frutas) => ({ frutas, total: frutas.reduce((s, f) => s + pesos[f], 0) });
  let ecuaciones;
  let pregunta;
  if (aciertos < 5) {
    ecuaciones = [ecuacion(...Array(entre(2, 4, rnd)).fill(A))];
    pregunta = A;
  } else if (aciertos < 10) {
    ecuaciones = [ecuacion(A, A), ecuacion(A, B)];
    pregunta = B;
  } else {
    ecuaciones = [ecuacion(A, A), ecuacion(A, B), ecuacion(B, C)];
    pregunta = C;
  }
  return { ecuaciones, pregunta, respuesta: pesos[pregunta], pesos };
}

export function start(pantalla, alTerminar, opciones = {}) {
  return jugarConOpciones(pantalla, alTerminar, (aciertos) => {
    const { ecuaciones, pregunta, respuesta } = generarBalanza(aciertos);
    const lineas = ecuaciones.map((e) => `⚖️ ${e.frutas.join(' ')} = ${e.total}`);
    return { texto: [...lineas, `${pregunta} = ?`].join('\n'), opciones: generarOpciones(respuesta), correcta: respuesta };
  }, 'balanza', opciones);
}
```

// Parejas: encuentra todas las parejas de cartas con los menos intentos posibles.
import { barajar, el } from './comun.js';
import { destello, sonar } from './efectos.js';

export const SIMBOLOS = ['🍎', '🐶', '⭐', '🌙', '🚗', '🎈', '🌵', '🐟', '🍕', '⚽'];
const PAREJAS = { facil: 4, normal: 6, dificil: 8 };

// n parejas al azar: cada símbolo dos veces, barajadas.
export function repartir(n, rnd = Math.random) {
  const elegidos = barajar(SIMBOLOS, rnd).slice(0, n);
  return barajar([...elegidos, ...elegidos], rnd);
}

export function start(pantalla, alTerminar, opciones = {}) {
  const { dificultad = 'normal' } = opciones;
  const cartas = repartir(PAREJAS[dificultad]);
  const total = cartas.length / 2;
  let abiertas = []; // cartas levantadas en este intento (0, 1 o 2)
  let encontradas = 0;
  let intentos = 0;
  let terminado = false;
  let id;

  const marcadorIntentos = el('span', '', 'Intentos: 0');
  const marcadorParejas = el('span', '', `Parejas: 0/${total}`);
  const marcador = el('div', 'marcador');
  marcador.append(marcadorIntentos, marcadorParejas);
  const mensaje = el('p', 'mensaje');
  const rejilla = el('div', 'parejas');
  const botones = cartas.map((_, i) => {
    const b = el('button', 'carta');
    b.ariaLabel = 'Carta boca abajo';
    b.onclick = () => tocar(i);
    return b;
  });
  rejilla.append(...botones);
  pantalla.append(marcador, mensaje, rejilla);

  function mostrar(i, visible) {
    botones[i].textContent = visible ? cartas[i] : '';
    botones[i].classList.toggle('vista', visible);
    botones[i].ariaLabel = visible ? cartas[i] : 'Carta boca abajo';
  }

  function tocar(i) {
    if (terminado || abiertas.length === 2 || abiertas.includes(i) || botones[i].classList.contains('hecha')) return;
    mostrar(i, true);
    sonar('tic');
    abiertas.push(i);
    if (abiertas.length < 2) return;
    intentos++;
    marcadorIntentos.textContent = `Intentos: ${intentos}`;
    const [a, b] = abiertas;
    if (cartas[a] === cartas[b]) {
      // Pareja: se quedan descubiertas (un solo sonido de acierto).
      botones[a].classList.add('hecha');
      botones[b].classList.add('hecha');
      destello(botones[a], true, false);
      destello(botones[b], true);
      abiertas = [];
      encontradas++;
      marcadorParejas.textContent = `Parejas: ${encontradas}/${total}`;
      if (encontradas === total) {
        terminado = true;
        mensaje.textContent = '✓ ¡Todas!';
        id = setTimeout(() => alTerminar(intentos), 700);
      }
    } else {
      // No son pareja: se ven un momento y se tapan.
      destello(botones[b], false);
      id = setTimeout(() => {
        for (const k of abiertas) mostrar(k, false);
        abiertas = [];
      }, 800);
    }
  }

  return () => {
    terminado = true;
    clearTimeout(id);
  };
}
```
