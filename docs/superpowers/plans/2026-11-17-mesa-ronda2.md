**Objetivo:** sumar cuatro juegos a 🎲 Mesa: Damas latinas, Reversi, Puntos y cajas y Batalla naval.
- Todos se juegan contra la máquina (Fácil, Normal, Difícil), a dos en un celular y contra un amigo.

- Las reglas de cada juego van en un módulo puro en `games/`, con la interfaz del Gato: `inicial`, `validas`, `aplicar`, `ganador`, `ia`.
- Cada juego se registra en `JUEGOS_MESA` de `app/mesa.js`, con su función `tablero(ctx)`. El tablero vive en `app/mesa-<id>.js`.
- La Tarea 1 generaliza el motor (`app/mesa.js`) con ganchos opcionales por juego. El resto de las tareas solo agrega juegos.

- Firebase Realtime Database para el modo amigo.

- Chrome sin ventana mediante el script CDP del scratchpad.

**Interfaces:**
- **Produce:** los ganchos opcionales de cada entrada de `JUEGOS_MESA`, que usarán las tareas siguientes.
  - `tablero(ctx) → HTMLElement`. Es **obligatorio** y tiene este contexto:
    ```
    ctx = { partida, j, g, puedo, jugar }
    ```
    - `partida`: el objeto de la partida abierta.
    - `j`: el juego.
    - `g`: `ganadorActual()`.
    - `puedo`: `puedoJugar()`.
    - `jugar(jugada)`: envía una jugada.
  - `opciones`: arreglo `[{ clave, nombre, valores: [[valor, texto], ...], defecto }]`. Se muestra en la pantalla de modo y se pasa a `inicial(opciones)`.
  - `ultimaJugada(antes, despues)`: devuelve la jugada del rival a partir de dos estados, para animarla. Si no existe, no hay animación.
  - `aNube(e)` y `deNube(estadoNube)`: convierten el estado para guardarlo en RTDB. Por defecto, la identidad.
  - `marcador(e)`: devuelve `{ x: texto, o: texto }`, que se muestra bajo el nombre de cada jugador.
  - `preparar(ctx2)` es **async** y lo usa la naval (Tarea 7). Se llama en `empezar` antes de crear el estado.
    ```
    ctx2 = { modo, contenedor }
    ```
    Devuelve opciones extra que se mezclan con las de `inicial`, o `null` si el jugador canceló.
- **`partida` gana dos campos:**
  - `opciones`: las del juego elegidas;
  - `escribiendo`: lo usa la Tarea 8.

const GANA = 100000;

function negamax(e, reglas, prof, alfa, beta) {
  const g = reglas.ganador(e);
  if (g) return g === 'empate' ? 0 : (g === e.turno ? GANA + prof : -GANA - prof);
  if (prof === 0) return reglas.valor(e);
  let mejor = -Infinity;
  for (const j of reglas.validas(e)) {
    const h = reglas.aplicar(e, j);
    const v = h.turno === e.turno ? negamax(h, reglas, prof - 1, alfa, beta) : -negamax(h, reglas, prof - 1, -beta, -alfa);
    if (v > mejor) mejor = v;
    if (mejor > alfa) alfa = mejor;
    if (alfa >= beta) break;
  }
  return mejor;
}

// La mejor jugada para e.turno mirando `profundidad` jugadas; entre las igual de buenas, una al azar.
export function mejorJugada(e, reglas, profundidad, rnd = Math.random) {
  const jugadas = reglas.validas(e);
  const valores = jugadas.map((j) => {
    const h = reglas.aplicar(e, j);
    return h.turno === e.turno ? negamax(h, reglas, profundidad - 1, -Infinity, Infinity) : -negamax(h, reglas, profundidad - 1, -Infinity, Infinity);
  });
  const max = Math.max(...valores);
  const mejores = jugadas.filter((_, k) => valores[k] === max);
  return mejores[azar(mejores.length, rnd)];
}

// los peones solo comen hacia adelante; corona quien termina la jugada en la última fila. Todo puro.

export const EMPATE_SIN_COMER = 30;
const DIRS = [[-1, -1], [-1, 1], [1, -1], [1, 1]];
const otro = (t) => (t === 'x' ? 'o' : 'x');
const dentro = (f, c) => f >= 0 && f < 8 && c >= 0 && c < 8;
const adelante = (t) => (t === 'x' ? -1 : 1);

export function inicial() {
  let celdas = '';
  for (let f = 0; f < 8; f++) for (let c = 0; c < 8; c++) celdas += (f + c) % 2 === 1 ? (f < 3 ? 'o' : f > 4 ? 'x' : '.') : '.';
  return { celdas, turno: 'x', sinComer: 0 };
}

// Secuencias de captura desde i (la pieza ya se quitó de su casilla de origen en `celdas`).
function saltos(celdas, i, pieza, comidas, camino, salida) {
  const t = pieza.toLowerCase();
  const dama = pieza !== t;
  const f = i >> 3, c = i & 7;
  let alguno = false;
  for (const [df, dc] of DIRS) {
    if (!dama && df !== adelante(t)) continue;
    let ff = f + df, cc = c + dc;
    if (dama) while (dentro(ff, cc) && celdas[ff * 8 + cc] === '.') { ff += df; cc += dc; }
    if (!dentro(ff, cc)) continue;
    const k = ff * 8 + cc;
    const v = celdas[k];
    if (v === '.' || v.toLowerCase() === t || comidas.includes(k)) continue;
    let lf = ff + df, lc = cc + dc;
    while (dentro(lf, lc) && celdas[lf * 8 + lc] === '.') {
      alguno = true;
      saltos(celdas, lf * 8 + lc, pieza, [...comidas, k], [...camino, lf * 8 + lc], salida);
      if (!dama) break;
      lf += df; lc += dc;
    }
  }
  if (!alguno && comidas.length) salida.push({ camino, comidas });
}

// Todas las jugadas legales: si se puede comer, solo las que más comen.
export function jugadas(e) {
  const capturas = [];
  const simples = [];
  for (let i = 0; i < 64; i++) {
    const p = e.celdas[i];
    if (p === '.' || p.toLowerCase() !== e.turno) continue;
    const sin = e.celdas.slice(0, i) + '.' + e.celdas.slice(i + 1);
    saltos(sin, i, p, [], [i], capturas);
    const dama = p !== e.turno;
    const f = i >> 3, c = i & 7;
    for (const [df, dc] of DIRS) {
      if (!dama && df !== adelante(e.turno)) continue;
      let ff = f + df, cc = c + dc;
      while (dentro(ff, cc) && e.celdas[ff * 8 + cc] === '.') {
        simples.push({ camino: [i, ff * 8 + cc], comidas: [] });
        if (!dama) break;
        ff += df; cc += dc;
      }
    }
  }
  if (!capturas.length) return simples;
  const max = Math.max(...capturas.map((j) => j.comidas.length));
  return capturas.filter((j) => j.comidas.length === max);
}

export const validas = (e) => [...new Set(jugadas(e).map((j) => j.camino.join('-')))];

export function aplicar(e, jugada) {
  const j = jugadas(e).find((x) => x.camino.join('-') === jugada);
  const celdas = e.celdas.split('');
  const desde = j.camino[0];
  const hasta = j.camino[j.camino.length - 1];
  let pieza = celdas[desde];
  const eraDama = pieza !== e.turno;
  celdas[desde] = '.';
  for (const k of j.comidas) celdas[k] = '.';
  if (!eraDama && (hasta >> 3) === (e.turno === 'x' ? 0 : 7)) pieza = pieza.toUpperCase();

  const sinComer = !j.comidas.length && eraDama ? e.sinComer + 1 : 0;
  return { celdas: celdas.join(''), turno: otro(e.turno), sinComer };
}

  if (!jugadas(e).length) return otro(e.turno);
  if (e.sinComer >= EMPATE_SIN_COMER) return 'empate';
  return null;
}

// Valor para quien tiene el turno: material (peón 1, dama 3) y, en la difícil, avance de los peones y centro.
function valorCon(posicion) {
  return (e) => {
    let v = 0;
    for (let i = 0; i < 64; i++) {
      const p = e.celdas[i];
      if (p === '.') continue;
      const t = p.toLowerCase();
      let m = p === t ? 1 : 3;
      if (posicion && p === t) m += 0.05 * (t === 'x' ? 7 - (i >> 3) : i >> 3);
      if (posicion && (i & 7) >= 2 && (i & 7) <= 5) m += 0.03;
      v += t === e.turno ? m : -m;
    }
    return v;
  };
}

const DIFICIL = { validas, aplicar, ganador, valor: valorCon(true) };

// ✏️ Puntos y cajas con n×n cajas (3, 4 o 5). Estado { n, celdas, turno }: celdas = líneas ('.' libre, '-' trazada)
// seguidas de los dueños de cada caja ('.', 'x', 'o'). Cerrar una caja da otro turno. Todo puro.
import { azar } from './comun.js';

const otro = (t) => (t === 'x' ? 'o' : 'x');
export const lineas = (n) => 2 * n * (n + 1);
const h = (n, f, c) => f * n + c;
const v = (n, f, c) => n * (n + 1) + f * (n + 1) + c;
export const ladosDe = (n, f, c) => [h(n, f, c), h(n, f + 1, c), v(n, f, c), v(n, f, c + 1)];

// Cajas (índice f*n+c) que toca la línea l.
function cajasDe(n, l) {
  const H = n * (n + 1);
  if (l < H) { const f = Math.floor(l / n), c = l % n; return [f - 1, f].filter((ff) => ff >= 0 && ff < n).map((ff) => ff * n + c); }
  const k = l - H, f = Math.floor(k / (n + 1)), c = k % (n + 1);
  return [c - 1, c].filter((cc) => cc >= 0 && cc < n).map((cc) => f * n + cc);
}
const ladosTrazados = (n, ls, b) => ladosDe(n, Math.floor(b / n), b % n).filter((l) => ls[l] === '-').length;

export function inicial({ n = 4 } = {}) {
  return { n, celdas: '.'.repeat(lineas(n) + n * n), turno: 'x' };
}

export const validas = (e) => [...Array(lineas(e.n)).keys()].filter((l) => e.celdas[l] === '.');

// Traza l sobre las líneas `ls` (texto): devuelve las líneas nuevas y cuántas cajas cerró.
function trazar(n, ls, l) {
  const nuevas = ls.slice(0, l) + '-' + ls.slice(l + 1);
  return { nuevas, cerradas: cajasDe(n, l).filter((b) => ladosTrazados(n, nuevas, b) === 4) };
}

export function aplicar(e, l) {
  const L = lineas(e.n);
  const { nuevas, cerradas } = trazar(e.n, e.celdas.slice(0, L), l);
  const duenos = e.celdas.slice(L).split('');
  for (const b of cerradas) duenos[b] = e.turno;
  return { n: e.n, celdas: nuevas + duenos.join(''), turno: cerradas.length ? e.turno : otro(e.turno) };
}

export function contar(e) {
  const d = e.celdas.slice(lineas(e.n));
  return { x: [...d].filter((c) => c === 'x').length, o: [...d].filter((c) => c === 'o').length };
}

export function ganador(e) {
  if (e.celdas.slice(0, lineas(e.n)).includes('.')) return null;
  const { x, o } = contar(e);
  return x === o ? 'empate' : x > o ? 'x' : 'o';
}

// Final exacto: mejor diferencia de cajas futuras (del que mueve menos las del rival) desde estas líneas.
function resolver(n, ls, memo) {
  if (!ls.includes('.')) return 0;
  if (memo.has(ls)) return memo.get(ls);
  let mejor = -Infinity;
  for (let l = 0; l < ls.length; l++) {
    if (ls[l] !== '.') continue;
    const { nuevas, cerradas } = trazar(n, ls, l);
    const v2 = cerradas.length ? cerradas.length + resolver(n, nuevas, memo) : -resolver(n, nuevas, memo);
    if (v2 > mejor) mejor = v2;
  }
  memo.set(ls, mejor);
  return mejor;
}
const FINAL_EXACTO = 14; // ponytail: con ≤ 14 líneas libres se resuelve exacto (2^14 estados); subir si el celular aguanta

// Cuántas cajas regala la línea l: el rival cierra todo lo que pueda en cadena.
function regala(n, ls, l) {
  let { nuevas } = trazar(n, ls, l);
  let total = 0;
  for (;;) {
    const libre = [...nuevas].findIndex((c, k) => c === '.' && trazar(n, nuevas, k).cerradas.length);
    if (libre < 0) return total;
    const r = trazar(n, nuevas, libre);
    total += r.cerradas.length;
    nuevas = r.nuevas;
  }
}

export function ia(e, nivel = 'normal', rnd = Math.random) {
  const n = e.n;
  const ls = e.celdas.slice(0, lineas(n));
  const libres = validas(e);
  const una = (arr) => arr[azar(arr.length, rnd)];
  if (nivel === 'facil') return una(libres);
  if (nivel === 'dificil' && libres.length <= FINAL_EXACTO) {
    const memo = new Map();
    const valores = libres.map((l) => { const { nuevas, cerradas } = trazar(n, ls, l); return cerradas.length ? cerradas.length + resolver(n, nuevas, memo) : -resolver(n, nuevas, memo); });
    const max = Math.max(...valores);
    return una(libres.filter((_, k) => valores[k] === max));
  }
  const cierran = libres.filter((l) => trazar(n, ls, l).cerradas.length);
  if (cierran.length) return una(cierran);
  const seguras = libres.filter((l) => cajasDe(n, l).every((b) => ladosTrazados(n, ls, b) < 2));
  if (seguras.length) return una(seguras);
  const costos = libres.map((l) => regala(n, ls, l));
  const min = Math.min(...costos);
  return una(libres.filter((_, k) => costos[k] === min));
}

// Barco: { c: casilla inicial, l: largo, v: vertical }. Estado: { turno, tiros: { x: [[casilla, resp]], o }, pendiente,
// barcos? (máquina y local), sellos? y revelado? (con un amigo) }. Respuestas: 'a' agua, 't' tocado, 'h<largo>' hundido.

import { azar } from './comun.js';

export const LADO = 10;
export const FLOTA = [5, 4, 3, 3, 2];
const otro = (t) => (t === 'x' ? 'o' : 'x');

export const casillasDe = ({ c, l, v }) => Array.from({ length: l }, (_, k) => c + k * (v ? LADO : 1));
const cabe = ({ c, l, v }) => Number.isInteger(c) && c >= 0 && c < LADO * LADO && (v ? Math.floor(c / LADO) + l <= LADO : (c % LADO) + l <= LADO);
function vecinas8(i) {
  const f = Math.floor(i / LADO), c = i % LADO, r = [];
  for (let df = -1; df <= 1; df++) for (let dc = -1; dc <= 1; dc++) {
    const ff = f + df, cc = c + dc;
    if ((df || dc) && ff >= 0 && ff < LADO && cc >= 0 && cc < LADO) r.push(ff * LADO + cc);
  }
  return r;
}

export function flotaValida(barcos) {
  if (!Array.isArray(barcos) || barcos.length !== FLOTA.length) return false;
  if (barcos.map((b) => b?.l).sort().join() !== [...FLOTA].sort().join()) return false;
  const ocupada = new Map();
  for (const [k, b] of barcos.entries()) {
    if (typeof b.v !== 'boolean' || !cabe(b)) return false;
    for (const i of casillasDe(b)) { if (ocupada.has(i)) return false; ocupada.set(i, k); }
  }
  for (const [i, k] of ocupada) for (const j of vecinas8(i)) if (ocupada.has(j) && ocupada.get(j) !== k) return false;
  return true;
}

export function flotaAlAzar(rnd = Math.random) {
  for (;;) {
    const barcos = [];
    for (const l of FLOTA) {
      let puesto = false;
      for (let intento = 0; intento < 200 && !puesto; intento++) {
        const b = { c: azar(LADO * LADO, rnd), l, v: rnd() < 0.5 };
        const prueba = [...barcos, b];
        if (cabe(b) && validaParcial(prueba)) { barcos.push(b); puesto = true; }
      }
      if (!puesto) break;
    }
    if (barcos.length === FLOTA.length) return barcos;
  }
}
// Como flotaValida pero sin exigir la flota completa (para ir colocando).
function validaParcial(barcos) {
  const ocupada = new Map();
  for (const [k, b] of barcos.entries()) for (const i of casillasDe(b)) { if (ocupada.has(i)) return false; ocupada.set(i, k); }
  for (const [i, k] of ocupada) for (const j of vecinas8(i)) if (ocupada.has(j) && ocupada.get(j) !== k) return false;
  return true;
}

export function girar(barcos, k) {
  const otra = barcos.map((b, i) => (i === k ? { c: b.c, l: b.l, v: !b.v } : b));
  return flotaValida(otra) ? otra : null;
}
export function mover(barcos, k, casilla) {
  const otra = barcos.map((b, i) => (i === k ? { c: casilla, l: b.l, v: b.v } : b));
  return flotaValida(otra) ? otra : null;
}

// Respuesta al tiro en `casilla`, sabiendo las casillas ya recibidas antes.
export function respuesta(barcos, recibidas, casilla) {
  const b = barcos.find((x) => casillasDe(x).includes(casilla));
  if (!b) return 'a';
  const tocadas = new Set([...recibidas, casilla]);
  return casillasDe(b).every((i) => tocadas.has(i)) ? 'h' + b.l : 't';
}

export function inicial({ barcos = null, amigo = false } = {}) {
  const e = { turno: amigo ? 'o' : 'x', tiros: { x: [], o: [] }, pendiente: null };
  if (barcos) e.barcos = barcos;
  if (amigo) { e.sellos = {}; e.revelado = {}; }
  return e;
}

export function validas(e) {
  if (e.pendiente !== null && e.pendiente !== undefined) return [];
  const hechos = new Set(e.tiros[e.turno].map((t) => t[0]));
  return [...Array(LADO * LADO).keys()].filter((i) => !hechos.has(i));
}

export function aplicar(e, casilla) {
  const rival = otro(e.turno);
  if (e.barcos) {
    const r = respuesta(e.barcos[rival], e.tiros[e.turno].map((t) => t[0]), casilla);
    return { ...e, tiros: { ...e.tiros, [e.turno]: [...e.tiros[e.turno], [casilla, r]] }, turno: rival };
  }
  return { ...e, pendiente: casilla, turno: rival };
}

// Con un amigo: quien tiene el turno responde al tiro pendiente del rival con su flota (el turno no cambia).
export function responder(e, misBarcos) {
  const tirador = otro(e.turno);
  const r = respuesta(misBarcos, e.tiros[tirador].map((t) => t[0]), e.pendiente);
  return { ...e, tiros: { ...e.tiros, [tirador]: [...e.tiros[tirador], [e.pendiente, r]] }, pendiente: null };
}

const hundidos = (tiros) => tiros.filter(([, r]) => r[0] === 'h').length;
export const vencedor = (e) => ['x', 'o'].find((j) => hundidos(e.tiros[j]) === FLOTA.length) ?? null;

export function ganador(e) {
  const v = vencedor(e);
  if (!v) return null;
  if (e.sellos && !(e.revelado?.x && e.revelado?.o)) return null;
  return v;
}

// ¿Las respuestas dadas a los tiros del rival son las que corresponden a esta flota?
export function coherente(barcos, tirosDelRival) {
  const recibidas = [];
  for (const [c, r] of tirosDelRival) {
    if (respuesta(barcos, recibidas, c) !== r) return false;
    recibidas.push(c);
  }
  return true;
}

export const nuevaSal = () => [...globalThis.crypto.getRandomValues(new Uint8Array(16))].map((b) => b.toString(16).padStart(2, '0')).join('');

export async function sellar(barcos, sal) {
  const datos = new TextEncoder().encode(JSON.stringify(barcos) + sal);
  const h = await globalThis.crypto.subtle.digest('SHA-256', datos);
  return [...new Uint8Array(h)].map((b) => b.toString(16).padStart(2, '0')).join('');
}

// ¿Lo que reveló `ficha` coincide con su sello y con todas las respuestas que dio?
export async function verificar(e, ficha) {
  const rv = e.revelado?.[ficha];
  if (!rv || !flotaValida(rv.barcos)) return false;
  if ((await sellar(rv.barcos, rv.sal)) !== e.sellos?.[ficha]) return false;
  return coherente(rv.barcos, e.tiros[otro(ficha)]);
}

// Casillas de barcos ya hundidos: el grupo de tocados unidos a cada 'h' (los barcos no se tocan, así que es exacto).
function analizar(tiros) {
  const resp = new Map(tiros);
  const tocados = new Set(tiros.filter(([, r]) => r !== 'a').map(([c]) => c));
  const hundidas = new Set();
  const largosHundidos = [];
  for (const [c, r] of tiros) {
    if (r[0] !== 'h') continue;
    largosHundidos.push(Number(r.slice(1)));
    const pila = [c];
    while (pila.length) {
      const i = pila.pop();
      if (hundidas.has(i)) continue;
      hundidas.add(i);
      for (const j of [i - LADO, i + LADO, i % LADO ? i - 1 : -1, i % LADO < LADO - 1 ? i + 1 : -1]) if (j >= 0 && j < 100 && tocados.has(j)) pila.push(j);
    }
  }
  const vivos = [...tocados].filter((i) => !hundidas.has(i));
  const restantes = [...FLOTA];
  for (const l of largosHundidos) restantes.splice(restantes.indexOf(l), 1);
  return { resp, hundidas, vivos, restantes };
}

export function ia(e, nivel = 'normal', rnd = Math.random) {
  const libres = validas(e);
  const una = (arr) => arr[azar(arr.length, rnd)];
  if (nivel === 'facil') return una(libres);
  const { resp, hundidas, vivos, restantes } = analizar(e.tiros[e.turno]);
  const libre = new Set(libres);
  // Alrededor de un barco hundido no puede haber otro.
  const prohibidas = new Set([...hundidas].flatMap(vecinas8));
  if (nivel === 'normal') {
    if (vivos.length) {
      const enLinea = vivos.length > 1 ? (vivos[1] - vivos[0]) % LADO === 0 ? [LADO] : [1] : [1, LADO];
      const cand = vivos.flatMap((i) => enLinea.flatMap((d) => [i - d, i + d]))
        .filter((j) => libre.has(j) && !prohibidas.has(j) && (Math.abs(j - vivos[0]) % LADO === 0 || Math.floor(j / LADO) === Math.floor(vivos[0] / LADO)));
      if (cand.length) return una(cand);
    }
    const paridad = libres.filter((i) => (Math.floor(i / LADO) + (i % LADO)) % 2 === 0 && !prohibidas.has(i));
    return una(paridad.length ? paridad : libres);
  }
  // Difícil: cuenta en cuántas posiciones posibles de los barcos restantes cae cada casilla libre.
  const peso = new Map();
  for (const l of new Set(restantes)) {
    for (let c = 0; c < 100; c++) for (const v of [false, true]) {
      const b = { c, l, v };
      if (!cabe(b)) continue;
      const cs = casillasDe(b);
      if (cs.some((i) => resp.get(i) === 'a' || hundidas.has(i) || prohibidas.has(i))) continue;
      const cubre = cs.filter((i) => vivos.includes(i)).length;

      for (const i of cs) if (libre.has(i)) peso.set(i, (peso.get(i) ?? 0) + (cubre ? 20 * cubre : 1));
    }
  }
  if (!peso.size) return una(libres);
  const max = Math.max(...peso.values());
  return una([...peso].filter(([, p]) => p === max).map(([i]) => i));
}
